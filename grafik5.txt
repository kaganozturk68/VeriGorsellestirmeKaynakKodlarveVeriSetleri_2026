library(readxl)
library(ggplot2)
library(scales)

options(scipen = 999)

veri5 <- read_excel("surekli_denetim_sayilari.xlsx")
names(veri5) <- c("Yil", "Denetim_Sayisi")

grafik5 <- ggplot(veri5, aes(x = Yil, y = Denetim_Sayisi)) +
  
  geom_line(
    linewidth = 1.5,
    color = "#2C3E50"
  ) +
  
  geom_point(
    size = 4,
    color = "#FF5733"
  ) +
  
  scale_y_continuous(
    labels = comma
  ) +
  
  scale_x_continuous(
    breaks = seq(
      min(veri5$Yil),
      max(veri5$Yil),
      by = 1
    )
  ) +
  
  labs(
    title = "Y\u0131llara G\u00f6re Denetim Say\u0131lar\u0131",
    x = "Y\u0131l",
    y = "Denetim Say\u0131s\u0131"
  ) +
  
  theme_minimal() +
  
  theme(
    plot.title = element_text(
      size = 24,
      face = "bold",
      hjust = 0.5
    ),
    
    axis.title = element_text(size = 16),
    
    axis.text = element_text(size = 12)
  )

ggsave(
  filename = "grafik5_linechart.png",
  plot = grafik5,
  width = 12,
  height = 7,
  dpi = 300
)

print(grafik5)
