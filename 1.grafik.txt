library(readxl)
library(tidyr)
library(ggplot2)
library(dplyr)

veri <- read_excel("kategorik_cinsiyet_yillara_gore.xlsx")

names(veri) <- tolower(trimws(names(veri)))

veri <- veri %>% select(-contains("toplam"))

yil <- names(veri)[1]

veri_long <- pivot_longer(
  veri,
  cols = -all_of(yil),
  names_to = "kategori",
  values_to = "oran"
)

grafik <- ggplot(
  veri_long,
  aes(
    x = .data[[yil]],
    y = oran,
    fill = kategori
  )
) +
  geom_bar(
    stat = "identity",
    position = "dodge"
  ) +
  scale_x_continuous(
    breaks = seq(min(veri[[yil]]), max(veri[[yil]]), by = 1)
  ) +
  labs(
    title = "Cinsiyete Gore Genclerde Sigara Kullanimi",
    x = "Yil",
    y = "Oran (%)",
    fill = "Kategori"
  ) +
  theme_minimal() +
  theme(
    plot.title = element_text(
      hjust = 0.5,
      size = 20,
      face = "bold"
    ),
    axis.title = element_text(size = 15),
    axis.text = element_text(size = 12),
    legend.title = element_text(size = 14),
    legend.text = element_text(size = 12)
  )

ggsave(
  filename = "grafik1_gruplu_sutun.png",
  plot = grafik,
  width = 8,
  height = 5,
  dpi = 300
)

print(grafik)
