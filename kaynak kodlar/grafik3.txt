library(readxl)
library(ggplot2)
library(dplyr)

veri <- read_excel("kategorik_tbm_modullerine_gore_egitim.xlsx", skip = 2)

names(veri) <- c("Modul", "Ogretmen", "Ogrenci", "Veli")

veri <- veri %>%
  filter(!is.na(Modul)) %>%
  mutate(
    Ogretmen = as.numeric(gsub("[^0-9]", "", Ogretmen)),
    Ogrenci = as.numeric(gsub("[^0-9]", "", Ogrenci)),
    Veli = as.numeric(gsub("[^0-9]", "", Veli))
  ) %>%
  filter(
    !is.na(Ogretmen),
    !is.na(Ogrenci),
    !is.na(Veli)
  ) %>%
  mutate(
    Toplam = Ogretmen + Ogrenci + Veli,
    Oran = round(Toplam / sum(Toplam) * 100, 1),
    Etiket = paste0(Oran, "%")
  )

grafik3 <- ggplot(veri, aes(x = "", y = Toplam, fill = Modul)) +

  geom_col(
    width = 1,
    color = "white",
    linewidth = 1
  ) +

  coord_polar(theta = "y") +

  geom_text(
    aes(label = Etiket),
    position = position_stack(vjust = 0.5),
    size = 5,
    color = "black",
    fontface = "bold"
  ) +

  labs(
    title = "TBM Mod\u00fcllerine G\u00f6re E\u011fitim Da\u011f\u0131l\u0131m\u0131",
    fill = "TBM Mod\u00fcl\u00fc"
  ) +

  theme_void() +

  theme(
    plot.background = element_rect(fill = "white", color = NA),
    panel.background = element_rect(fill = "white", color = NA),

    plot.title = element_text(
      size = 22,
      face = "bold",
      hjust = 0.5
    ),

    legend.title = element_text(
      size = 13,
      face = "bold"
    ),

    legend.text = element_text(
      size = 12
    ),

    legend.position = "right"
  )

ggsave(
  filename = "grafik3_pasta.png",
  plot = grafik3,
  width = 10,
  height = 7,
  dpi = 300,
  bg = "white"
)
print(grafik3)
