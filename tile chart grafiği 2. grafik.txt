library(ggplot2)

Sys.setlocale("LC_ALL", "Turkish")

veri <- data.frame(
  Hizmet_Turu = c(
    "Poliklinik","Poliklinik",
    "ALO 171","ALO 171","ALO 171",
    "Ilac","Ilac",
    "TUBATIS","TUBATIS",
    "Geri Arama","Geri Arama"
  ),
  Ozellik = c(
    "Yuz yuze","Hekim",
    "Telefon","24 saat","4000 cagri",
    "Nikotin","Recete",
    "Dijital","Veri",
    "6 kez","Motivasyon"
  )
)

grafik <- ggplot(
  veri,
  aes(
    x = Hizmet_Turu,
    y = Ozellik,
    fill = Hizmet_Turu
  )
) +
  geom_tile(
    color = "white",
    linewidth = 2,
    width = 0.9,
    height = 0.9
  ) +
  geom_text(
    label = "✓",
    color = "white",
    size = 11,
    fontface = "bold"
  ) +
  labs(
    title = "Sigara Bırakma Hizmetleri ve Özellikleri",
    subtitle = "Hizmet türleri ile özelliklerin eşleşme matrisi",
    x = "Hizmet Türü",
    y = "Özellik",
    fill = "Hizmet Türü"
  ) +
  theme_minimal() +
  theme(
    plot.background = element_rect(fill = "white", color = "white"),
    panel.background = element_rect(fill = "white", color = "white"),
    plot.title = element_text(size = 26, face = "bold", hjust = 0.5),
    plot.subtitle = element_text(size = 15, hjust = 0.5, color = "gray30"),
    axis.title = element_text(size = 15, face = "bold"),
    axis.text.x = element_text(size = 13, face = "bold", angle = 25, hjust = 1),
    axis.text.y = element_text(size = 13, face = "bold"),
    panel.grid = element_blank(),
    legend.position = "none"
  )

print(grafik)

ggsave(
  filename = "2. grafik tile chart grafiği.png",
  plot = grafik,
  width = 14,
  height = 9,
  dpi = 300,
  bg = "white"
)
